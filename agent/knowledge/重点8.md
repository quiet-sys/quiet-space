## 二维数组

整数类型

讲解二维数组遍历 == 图形打印

```java
package test;

public class Test3 {
	
	public static void main(String[] args) {
		  //0		1	2	3	4	5	6	7	8
		//0 1
		//1			1
		//2				1
		//3					1
		//4						1
		//5							1
		//6								1
		//7									1
		//8										1
		int [][] num = new int[9][9];
		//int [][] num = {{1,2,3},{5,6,7},{9,10,11}};
		num[0][0] = 1;
		num[1][1] = 1;
		num[2][2] = 1;
		num[3][3] = 1;
		num[4][4] = 1;
		num[5][5] = 1;
		num[6][6] = 1;
		num[7][7] = 1;
		num[8][8] = 1;
		
		/*System.out.println(num.length);
		System.out.println(num[0].length);*/
		
		//二维使用还是只能使用下标
		//遍历
		for(int i = 0;i<num.length;i++) {//行
			for(int j = 0;j<num[0].length;j++) {//列
				if(num[i][j]==0) {
					System.out.print(" ");
				}else {
					System.out.print(num[i][j]+"  ");
				}
				}
				
			System.out.println();
		}
	}

}

```
